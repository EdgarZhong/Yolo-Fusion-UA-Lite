# 任务 A：特征紧凑度与悖论机制分析

## A.0 目标

回答核心问题：**为什么特征可分性下降，检测性能反而上升？**

假说：注意力通过隐式降噪使同类目标的特征更紧凑（类内方差降低），代价是部分判别通道也被压缩（类间距离略降）。如果类内紧凑度的改善幅度大于类间距离的损失，检测头就更容易工作——因为检测头需要的不是"最大化边界"而是"在噪声条件下稳定分类"。

## A.1 输入

四个模型的现有 npz 文件 + M5 的 npz（共 5 个）。路径：`src/diagnostic/outputs/features/`。**不需要 GPU，不需要训练，纯 CPU numpy 计算。**

## A.2 需要实现的脚本

创建 `src/diagnostic/analyze/feature_compactness.py`，对每个模型的每个 feat_\* 键、每个类别计算以下指标：

### A.2.1 指标定义

**指标 1：类内特征方差（Intra-class Variance）**

对类别 c 的所有目标特征向量 {x_i}，计算每个通道的方差，再取均值：

```
intra_var(c) = mean_over_channels( var({x_i[k] : label_i == c}) )
```

即先对每个通道 k 求该类所有样本在通道 k 上的方差，再对所有通道取均值。值越小表示特征越紧凑。

**指标 2：类内余弦相似度（Intra-class Cosine Similarity）**

对类别 c，随机采样 N_pairs=10000 对同类目标对，计算余弦相似度的均值和标准差：

```
cos_sim_intra(c) = mean( cosine_similarity(x_i, x_j) )  for i,j both in class c
```

值越高表示同类特征越一致。

**指标 3：类间质心距离（Inter-class Centroid Distance）**

对类别对 (a, b)，计算两个类别的特征质心的欧氏距离和余弦距离：

```
centroid_a = mean({x_i : label_i == a})
centroid_b = mean({x_i : label_i == b})
inter_dist_L2(a,b) = ||centroid_a - centroid_b||_2
inter_dist_cos(a,b) = 1 - cosine_similarity(centroid_a, centroid_b)
```

**指标 4：判别信噪比（Discriminative SNR）**

```
SNR(a,b) = inter_dist_L2(a,b) / sqrt( intra_var(a) + intra_var(b) )
```

这是 Fisher 判别比的宏观版本。如果注意力降低了类间距离但同时大幅降低了类内方差，SNR 可能不降反升——这就能解释悖论。

**指标 5：特征范数统计（Feature Norm）**

```
norm_mean(c) = mean( ||x_i||_2 : label_i == c )
norm_std(c) = std( ||x_i||_2 : label_i == c )
```

注意力可能在归一化特征幅度，使检测头的 weight 更容易适配。

### A.2.2 完整脚本实现

```python
"""特征紧凑度分析脚本
用法：python feature_compactness.py --features <npz> --class-pairs "van:car" ... --output <json>
"""
import argparse, json, time
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity as cos_sim

CLASS_MAP = {"car": 0, "truck": 1, "bus": 2, "van": 3, "freight_car": 4}
N_COS_PAIRS = 10000  # 余弦相似度采样对数

def analyze_compactness(features, labels, cls_a_name, cls_b_name, seed=42):
    id_a, id_b = CLASS_MAP[cls_a_name], CLASS_MAP[cls_b_name]
    rng = np.random.RandomState(seed)
    
    feat_a = features[labels == id_a]
    feat_b = features[labels == id_b]
    
    # --- 指标 1：类内方差 ---
    intra_var_a = float(np.mean(np.var(feat_a, axis=0)))
    intra_var_b = float(np.mean(np.var(feat_b, axis=0)))
    
    # --- 指标 2：类内余弦相似度（采样） ---
    def sample_cos_intra(feat, n_pairs):
        n = len(feat)
        if n < 2:
            return float('nan'), float('nan')
        idx_i = rng.randint(0, n, n_pairs)
        idx_j = rng.randint(0, n, n_pairs)
        # 排除自身对
        mask = idx_i != idx_j
        idx_i, idx_j = idx_i[mask], idx_j[mask]
        sims = np.sum(feat[idx_i] * feat[idx_j], axis=1) / (
            np.linalg.norm(feat[idx_i], axis=1) * np.linalg.norm(feat[idx_j], axis=1) + 1e-8
        )
        return float(np.mean(sims)), float(np.std(sims))
    
    cos_intra_a_mean, cos_intra_a_std = sample_cos_intra(feat_a, N_COS_PAIRS)
    cos_intra_b_mean, cos_intra_b_std = sample_cos_intra(feat_b, N_COS_PAIRS)
    
    # --- 指标 3：类间质心距离 ---
    centroid_a = np.mean(feat_a, axis=0)
    centroid_b = np.mean(feat_b, axis=0)
    inter_dist_l2 = float(np.linalg.norm(centroid_a - centroid_b))
    cos_val = np.dot(centroid_a, centroid_b) / (
        np.linalg.norm(centroid_a) * np.linalg.norm(centroid_b) + 1e-8
    )
    inter_dist_cos = float(1.0 - cos_val)
    
    # --- 指标 4：判别 SNR ---
    snr = inter_dist_l2 / (np.sqrt(intra_var_a + intra_var_b) + 1e-8)
    
    # --- 指标 5：特征范数 ---
    norms_a = np.linalg.norm(feat_a, axis=1)
    norms_b = np.linalg.norm(feat_b, axis=1)
    
    return {
        "class_pair": f"{cls_a_name}:{cls_b_name}",
        "n_a": int(len(feat_a)), "n_b": int(len(feat_b)),
        # 类内方差
        "intra_var_a": round(intra_var_a, 6),
        "intra_var_b": round(intra_var_b, 6),
        # 类内余弦
        "cos_intra_a_mean": round(cos_intra_a_mean, 6),
        "cos_intra_a_std": round(cos_intra_a_std, 6),
        "cos_intra_b_mean": round(cos_intra_b_mean, 6),
        "cos_intra_b_std": round(cos_intra_b_std, 6),
        # 类间距离
        "inter_dist_l2": round(inter_dist_l2, 6),
        "inter_dist_cos": round(inter_dist_cos, 6),
        # SNR
        "discriminative_snr": round(float(snr), 6),
        # 范数
        "norm_mean_a": round(float(np.mean(norms_a)), 6),
        "norm_std_a": round(float(np.std(norms_a)), 6),
        "norm_mean_b": round(float(np.mean(norms_b)), 6),
        "norm_std_b": round(float(np.std(norms_b)), 6),
    }

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--features", required=True)
    parser.add_argument("--class-pairs", nargs="+", required=True)
    parser.add_argument("--output", required=True)
    args = parser.parse_args()
    
    d = np.load(args.features, allow_pickle=True)
    labels = d["labels"]
    feat_keys = sorted([k for k in d.keys() if k.startswith("feat_")])
    
    results = []
    for fk in feat_keys:
        feat = d[fk]
        for pair_str in args.class_pairs:
            cls_a, cls_b = pair_str.split(":")
            r = analyze_compactness(feat, labels, cls_a, cls_b)
            r["feature_key"] = fk
            results.append(r)
            print(f"  {fk} | {pair_str}: intra_var={r['intra_var_a']:.4f}/{r['intra_var_b']:.4f}, "
                  f"SNR={r['discriminative_snr']:.4f}, cos_intra={r['cos_intra_a_mean']:.4f}/{r['cos_intra_b_mean']:.4f}")
    
    with open(args.output, "w") as f:
        json.dump(results, f, indent=2)
    print(f"Saved: {args.output}")

if __name__ == "__main__":
    main()
```

# 任务 B：光照条件分层诊断

## B.0 目标

DroneVehicle 包含白天和夜间场景。夜间 RGB 几乎无信息（全黑），双模态增益主要来自 IR。分层分析可以回答：**悖论是在所有光照条件下都存在，还是集中在特定条件下？**

如果悖论主要出现在夜间（RGB 无用时），说明注意力的降噪作用在单模态信息占主导时更重要。如果白天也存在，说明是普遍现象。

这是论文 Analysis 节的加分项——审稿人经常要求"在不同条件下验证你的结论"。

## B.1 输入

同任务 A：四个模型 + M5 的 npz，其中包含 `image_brightness` 字段。纯 CPU。

## B.2 需要实现的脚本

创建 `src/diagnostic/analyze/stratified_analysis.py`：

```python
"""光照分层诊断脚本
用法：python stratified_analysis.py --features <npz> --class-pairs "van:car" ... --brightness-threshold 40.0 --output <json>
"""
import argparse, json
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import StratifiedKFold

CLASS_MAP = {"car": 0, "truck": 1, "bus": 2, "van": 3, "freight_car": 4}

def linear_probe_balanced(feat, labels, id_a, id_b, seed=42):
    """配平后 5 折线性探针（复用现有逻辑）"""
    mask = np.isin(labels, [id_a, id_b])
    X, y = feat[mask], labels[mask]
    rng = np.random.RandomState(seed)
    n_min = min(np.sum(y == id_a), np.sum(y == id_b))
    if n_min < 20:
        return float('nan'), float('nan'), 0
    
    accs = []
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=seed)
    for train_idx, test_idx in skf.split(X, y):
        X_tr, y_tr = X[train_idx], y[train_idx]
        X_te, y_te = X[test_idx], y[test_idx]
        # 折内配平
        for split_X, split_y in [(X_tr, y_tr)]:
            n_min_fold = min(np.sum(split_y == id_a), np.sum(split_y == id_b))
            idx_a = rng.choice(np.where(split_y == id_a)[0], n_min_fold, replace=False)
            idx_b = rng.choice(np.where(split_y == id_b)[0], n_min_fold, replace=False)
            bal_idx = np.concatenate([idx_a, idx_b])
            X_tr, y_tr = split_X[bal_idx], split_y[bal_idx]
        clf = LogisticRegression(max_iter=1000, solver='lbfgs', random_state=seed)
        clf.fit(X_tr, y_tr)
        accs.append(clf.score(X_te, y_te))
    return float(np.mean(accs)), float(np.std(accs)), int(n_min * 2)

def fisher_top10(feat, labels, id_a, id_b, seed=42):
    """配平后 Fisher top-10 均值"""
    mask = np.isin(labels, [id_a, id_b])
    X, y = feat[mask], labels[mask]
    rng = np.random.RandomState(seed)
    n_min = min(np.sum(y == id_a), np.sum(y == id_b))
    if n_min < 20:
        return float('nan'), 0
    idx_a = rng.choice(np.where(y == id_a)[0], n_min, replace=False)
    idx_b = rng.choice(np.where(y == id_b)[0], n_min, replace=False)
    Xa, Xb = X[idx_a], X[idx_b]
    mu_a, mu_b = Xa.mean(0), Xb.mean(0)
    var_a, var_b = Xa.var(0), Xb.var(0)
    fisher = (mu_a - mu_b) ** 2 / (var_a + var_b + 1e-8)
    top10 = np.sort(fisher)[-10:]
    return float(np.mean(top10)), int(n_min * 2)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--features", required=True)
    parser.add_argument("--class-pairs", nargs="+", required=True)
    parser.add_argument("--brightness-threshold", type=float, default=40.0)
    parser.add_argument("--output", required=True)
    args = parser.parse_args()
    
    d = np.load(args.features, allow_pickle=True)
    labels = d["labels"]
    brightness = d["image_brightness"]
    feat_key = "feat_fused_p3"  # 只分析融合层
    feat = d[feat_key]
    
    threshold = args.brightness_threshold
    bright_mask = brightness >= threshold
    dark_mask = brightness < threshold
    
    print(f"Total: {len(labels)}, Bright(>={threshold}): {bright_mask.sum()}, Dark(<{threshold}): {dark_mask.sum()}")
    
    results = []
    for condition, mask in [("bright", bright_mask), ("dark", dark_mask), ("all", np.ones(len(labels), dtype=bool))]:
        feat_sub = feat[mask]
        labels_sub = labels[mask]
        for pair_str in args.class_pairs:
            cls_a, cls_b = pair_str.split(":")
            id_a, id_b = CLASS_MAP[cls_a], CLASS_MAP[cls_b]
            
            acc_mean, acc_std, n_probe = linear_probe_balanced(feat_sub, labels_sub, id_a, id_b)
            fisher_val, n_fisher = fisher_top10(feat_sub, labels_sub, id_a, id_b)
            
            results.append({
                "condition": condition,
                "class_pair": pair_str,
                "feature_key": feat_key,
                "linear_probe_acc": round(acc_mean, 4),
                "linear_probe_std": round(acc_std, 4),
                "fisher_top10": round(fisher_val, 4) if not np.isnan(fisher_val) else None,
                "n_samples": n_probe,
                "n_bright": int(bright_mask.sum()),
                "n_dark": int(dark_mask.sum()),
            })
            print(f"  {condition} | {pair_str}: probe={acc_mean:.4f}±{acc_std:.4f}, fisher={fisher_val:.4f}")
    
    with open(args.output, "w") as f:
        json.dump(results, f, indent=2)
    print(f"Saved: {args.output}")

if __name__ == "__main__":
    main()
```
